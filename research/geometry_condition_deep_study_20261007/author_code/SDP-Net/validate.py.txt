import csv

from torch import nn
from utils import *
import cv2
import pdb
from metrics import calc_psnr, calc_rmse, calc_ergas, calc_sam, calc_cc, calc_moae, calc_uiqi, calc_ssim
import args_parser
args = args_parser.args_parser()

def validate(test_list, arch, model, epoch, n_epochs):
    test_ref, test_lr, test_hr = test_list
    model.eval()

    psnr = 0
    with torch.no_grad():
        # Set mini-batch dataset
        ref = to_var(test_ref).detach()
        lr = to_var(test_lr).detach()
        hr = to_var(test_hr).detach()
        if arch == 'SSRNet':
            out, _, _, _, _, _ = model(lr, hr)
        elif arch == 'SSRSpat':
            _, out, _, _, _, _ = model(lr, hr)
        elif arch == 'SSRSpec':
            _, _, out, _, _, _ = model(lr, hr)
        elif arch == 'SwinCGAN':
            out, _, _, _, _, _ = model(lr, hr, args.scale_ratio)
        elif args.arch == 'TESTNet':
            out, _, _, _, _, _ = model(lr.cuda(), hr.cuda(), epoch)
        else:
            out, _, _, _, _, _ = model(lr, hr, args.scale_ratio)

        ref = ref.detach().cpu().numpy()
        out = out.detach().cpu().numpy()

        rmse = calc_rmse(ref, out)
        psnr = calc_psnr(ref, out)
        # ergas = compute_ergas(ref, out, args.scale_ratio)
        ergas = calc_ergas(ref, out)
        sam = calc_sam(ref, out)
        cc = calc_cc(ref, out)
        moae = calc_moae(ref, out)
        uiqi = calc_uiqi(ref, out)
        ssim = calc_ssim(ref, out)

    return psnr,rmse,ergas,sam,cc,moae,uiqi,ssim
