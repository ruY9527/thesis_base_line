import argparse

def args_parser():
    parser = argparse.ArgumentParser()
    parser.add_argument('-arch', type=str, default='SDPNet',
                        choices=[
                            'SDPNet',  # 2025
                        ])

    parser.add_argument('-root', type=str, default='./data')
    parser.add_argument('-dataset', type=str, default='chikusei',
                            choices=['chikusei'])
    parser.add_argument('--model_path', type=str, default='./checkpoints/dataset_arch_16.pkl',
                        help='path for trained encoder')
    parser.add_argument('--scale_ratio', type=float, default=4)

    parser.add_argument('--n_bands', type=int, default=128)
    parser.add_argument('--n_select_bands', type=int, default=4)

    parser.add_argument('--train_dir', type=str, default='./data/dataset/train',
                        help='directory for resized images')
    parser.add_argument('--val_dir', type=str, default='./data/dataset/val',
                        help='directory for resized images')

    # learning setting
    parser.add_argument('--n_epochs', type=int, default=9999,
                        help='end epoch for training')
    parser.add_argument('--lr', type=float, default=1e-4)
    parser.add_argument('--image_size', type=int, default=172)

    args = parser.parse_args()
    return args
